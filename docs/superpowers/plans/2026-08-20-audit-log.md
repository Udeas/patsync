# Audit Log Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Give admins a queryable, admin-only trail of what changed in the system — data create/update/delete, status changes, and auth events — with per-field diffs, backed by a new API and an Angular admin page.

**Architecture:** Backend captures data mutations automatically via a SQLAlchemy `before_flush`/`after_flush` listener on the `sqlmodel.Session` class (diff computed in before_flush while attribute history is live; rows Core-inserted in after_flush once PKs are assigned, in the same transaction). Status changes and auth events are recorded with explicit calls; a per-session marker de-dups a status change against the listener. An admin-only `GET /api/audit` (filter + paginate) reads the rows; an Angular `/admin/audit` page renders them.

**Tech Stack:** Python 3.13, FastAPI, SQLModel/SQLAlchemy, pytest (backend, `patsync/backend`); Angular 20, TypeScript, vitest (frontend).

## Global Constraints

- Backend repo: `patsync` on branch `main`; run all python/pytest/git from `patsync/backend`. Frontend repo: `frontend` on branch `feat/deadlines-page`; run from `frontend` (`npm test`).
- DB schema changes go in BOTH the Postgres and SQLite branches of `app/database.py`, idempotent (`CREATE TABLE IF NOT EXISTS` + `CREATE INDEX IF NOT EXISTS`), mirroring `_run_users_migration`. Never rewrite existing migration blocks. Backend tests build schema with `SQLModel.metadata.create_all`, so the model alone suffices in tests.
- Reuse the EXISTING `require_admin` dependency in `app/auth/deps.py` (do NOT add a new `get_current_admin`).
- `action` ∈ `{create, update, delete, status_change, login, login_failed, user_create}`. (Spec's `user_deactivate` is dropped — no deactivate endpoint exists.)
- `entity_type` ∈ `{patent, design, trademark, client, agent, user}`.
- `changes` is a JSON string: array of `{field, old, new}`; values JSON-coerced (dates → ISO, Decimal → str). Empty `[]` for create/delete/auth.
- Ignore-list fields (never diffed): `created_date`, `modified_date`, `last_status_updated_at`, `created_at`, `updated_at`.
- Audited parent models (allow-list) → `(entity_type, label attr)`: `PatentProject`→(`patent`,`docket_no`), `ApplicationData`→(`design`,`project_code`), `TmApplicationData`→(`trademark`,`project_code`), `PatentClient`→(`client`,`name`), `PatentAgent`→(`agent`,`name`). Every other model is ignored by the listener.
- Known pre-existing, out-of-scope test failures: `tests/test_health.py` (google.oauth2 import — may `--ignore`), `test_patent_project_collections.py` (one failure). Frontend: `patents-create.component.spec.ts`.

```python
from __future__ import annotations

import contextvars
from dataclasses import dataclass
from typing import Optional


@dataclass(frozen=True)
class AuditActor:
    user_id: Optional[int]
    username: Optional[str]


_current_actor: contextvars.ContextVar[Optional[AuditActor]] = contextvars.ContextVar(
    "audit_actor", default=None
)


def set_actor(user_id: Optional[int], username: Optional[str]) -> None:
    _current_actor.set(AuditActor(user_id=user_id, username=username))


def get_actor() -> Optional[AuditActor]:
    return _current_actor.get()


_MARKER_KEY = "_audit_explicit"


def mark_explicit(session, entity_type: str, entity_id: Optional[int]) -> None:
    marker = session.info.setdefault(_MARKER_KEY, set())
    marker.add((entity_type, entity_id))


def is_explicit(session, entity_type: str, entity_id: Optional[int]) -> bool:
    marker = session.info.get(_MARKER_KEY)
    return bool(marker) and (entity_type, entity_id) in marker
```

```python
from __future__ import annotations

import json
from datetime import date, datetime
from decimal import Decimal
from typing import Optional

from sqlalchemy import event, inspect
from sqlmodel import Session

from app.audit.context import get_actor, is_explicit
from app.auth.models import AuditLog
from app.models.applications import ApplicationData
from app.models.trademark import TmApplicationData
from app.patents.models import PatentAgent, PatentClient, PatentProject

# model -> (entity_type, label attribute)
_AUDITED = {
    PatentProject: ("patent", "docket_no"),
    ApplicationData: ("design", "project_code"),
    TmApplicationData: ("trademark", "project_code"),
    PatentClient: ("client", "name"),
    PatentAgent: ("agent", "name"),
}
_IGNORE = {"created_date", "modified_date", "last_status_updated_at", "created_at", "updated_at"}
_UPDATE_SENTINELS = {"modified_date", "updated_at"}  # presence => an intentional update


def _coerce(value):
    if isinstance(value, (date, datetime)):
        return value.isoformat()
    if isinstance(value, Decimal):
        return str(value)
    if value is None or isinstance(value, (str, int, float, bool)):
        return value
    return str(value)


def _label(obj, attr: str) -> str:
    return str(getattr(obj, attr, None) or "")


def _identity(obj) -> Optional[int]:
    ident = inspect(obj).identity
    return ident[0] if ident else None


def _scalar_diff(obj):
    changes = []
    for attr in inspect(obj).attrs:
        if attr.key in _IGNORE:
            continue
        hist = attr.history
        if not hist.has_changes():
            continue
        old = hist.deleted[0] if hist.deleted else None
        new = hist.added[0] if hist.added else None
        changes.append({"field": attr.key, "old": _coerce(old), "new": _coerce(new)})
    return changes


def _update_touched(obj) -> bool:
    return any(inspect(obj).attrs[k].history.has_changes() for k in _UPDATE_SENTINELS if k in inspect(obj).attrs)


def _before_flush(session, flush_context, instances):
    pending = session.info.setdefault("_audit_pending", [])

    for obj in session.new:
        meta = _AUDITED.get(type(obj))
        if not meta:
            continue
        etype, label_attr = meta
        pending.append({"obj": obj, "action": "create", "entity_type": etype,
                        "label_attr": label_attr, "entity_id": None, "changes": [], "label": None})

    for obj in session.dirty:
        meta = _AUDITED.get(type(obj))
        if not meta:
            continue
        etype, label_attr = meta
        eid = _identity(obj)
        if is_explicit(session, etype, eid):
            continue
        changes = _scalar_diff(obj)
        if not changes and not _update_touched(obj):
            continue
        pending.append({"obj": obj, "action": "update", "entity_type": etype,
                        "label_attr": label_attr, "entity_id": eid, "changes": changes, "label": None})

    for obj in session.deleted:
        meta = _AUDITED.get(type(obj))
        if not meta:
            continue
        etype, label_attr = meta
        pending.append({"obj": None, "action": "delete", "entity_type": etype,
                        "label_attr": None, "entity_id": _identity(obj), "changes": [],
                        "label": _label(obj, label_attr)})


def _after_flush(session, flush_context):
    pending = session.info.get("_audit_pending")
    if not pending:
        return
    session.info["_audit_pending"] = []
    actor = get_actor()
    rows = []
    for p in pending:
        obj = p["obj"]
        if obj is not None:
            entity_id = getattr(obj, "id", None)
            label = _label(obj, p["label_attr"]) if p["label_attr"] else ""
        else:
            entity_id = p["entity_id"]
            label = p["label"] or ""
        rows.append({
            "created_at": datetime.utcnow(),
            "actor_user_id": actor.user_id if actor else None,
            "actor_username": actor.username if actor else None,
            "action": p["action"],
            "entity_type": p["entity_type"],
            "entity_id": entity_id,
            "entity_label": label,
            "changes": json.dumps(p["changes"]),
            "ip_address": None,
        })
    if rows:
        session.connection().execute(AuditLog.__table__.insert(), rows)


def register_audit_listener() -> None:
    if not event.contains(Session, "before_flush", _before_flush):
        event.listen(Session, "before_flush", _before_flush)
    if not event.contains(Session, "after_flush", _after_flush):
        event.listen(Session, "after_flush", _after_flush)
```

```python
        entity_type=entity_type,
        entity_id=entity_id,
        entity_label=entity_label,
        changes=json.dumps(changes or []),
        ip_address=ip_address,
    ))


def record_status_change(session, *, entity_type: str, entity_id: int, entity_label: str,
                         old_status: Optional[str], new_status: str,
                         extra_changes: Optional[list] = None) -> None:
    mark_explicit(session, entity_type, entity_id)
    changes = [{"field": "status", "old": old_status, "new": new_status}]
    if extra_changes:
        changes.extend(extra_changes)
    write_audit(session, action="status_change", entity_type=entity_type,
                entity_id=entity_id, entity_label=entity_label, changes=changes)
```

```python
from __future__ import annotations

import json
from datetime import date, datetime, time
from typing import Optional

from fastapi import APIRouter, Depends, Query
from sqlalchemy import func
from sqlmodel import Session, select

from app.audit.schemas import AuditListResponse, AuditLogRead
from app.auth.models import AuditLog
from app.database import get_session

router = APIRouter()


def _to_read(row: AuditLog) -> AuditLogRead:
    try:
        parsed = json.loads(row.changes) if row.changes else []
    except (ValueError, TypeError):
        parsed = []
    return AuditLogRead(
        id=row.id or 0,
        created_at=row.created_at,
        actor_username=row.actor_username,
        action=row.action,
        entity_type=row.entity_type,
        entity_id=row.entity_id,
        entity_label=row.entity_label,
        changes=parsed,
        ip_address=row.ip_address,
    )


@router.get("", response_model=AuditListResponse)
def list_audit(
    page: int = Query(default=1, ge=1),
    page_size: int = Query(default=25, ge=1, le=100),
    actor: Optional[str] = None,
    entity_type: Optional[str] = None,
    action: Optional[str] = None,
    date_from: Optional[date] = None,
    date_to: Optional[date] = None,
    session: Session = Depends(get_session),
):
    filters = []
    if actor:
        filters.append(func.lower(AuditLog.actor_username).like(f"%{actor.lower()}%"))
    if entity_type:
        filters.append(AuditLog.entity_type == entity_type)
    if action:
        filters.append(AuditLog.action == action)
    if date_from:
        filters.append(AuditLog.created_at >= datetime.combine(date_from, time.min))
    if date_to:
        filters.append(AuditLog.created_at <= datetime.combine(date_to, time.max))

    count_stmt = select(func.count()).select_from(AuditLog)
    for f in filters:
        count_stmt = count_stmt.where(f)
    total = session.exec(count_stmt).one()

    stmt = select(AuditLog)
    for f in filters:
        stmt = stmt.where(f)
    stmt = stmt.order_by(AuditLog.created_at.desc(), AuditLog.id.desc())
    stmt = stmt.offset((page - 1) * page_size).limit(page_size)
    rows = session.exec(stmt).all()

    return AuditListResponse(
        items=[_to_read(r) for r in rows],
        total=int(total),
        page=page,
        page_size=page_size,
    )
```

```ts
import { HttpClient, HttpParams } from '@angular/common/http';
import { Component, computed, inject, signal } from '@angular/core';
import { FormsModule } from '@angular/forms';
import { AuditChange, AuditListResponse, AuditLogRow } from './audit-types';

@Component({
  selector: 'app-audit-log',
  imports: [CommonModule, FormsModule],
  templateUrl: './audit-log.component.html',
  styleUrl: './audit-log.component.css'
})
export class AuditLogComponent {
  private readonly http = inject(HttpClient);
  private readonly apiUrl = `${environment.apiBaseUrl}/api/audit`;

  protected readonly rows = signal<AuditLogRow[]>([]);
  protected readonly total = signal(0);
  protected readonly page = signal(1);
  protected readonly pageSize = signal(25);
  protected readonly isLoading = signal(false);
  protected readonly loadError = signal('');

  protected readonly actor = signal('');
  protected readonly entityType = signal('');
  protected readonly action = signal('');
  protected readonly dateFrom = signal('');
  protected readonly dateTo = signal('');

  protected readonly entityTypes = ['patent', 'design', 'trademark', 'client', 'agent', 'user'];
  protected readonly actions = ['create', 'update', 'delete', 'status_change', 'login', 'login_failed', 'user_create'];

  protected readonly totalPages = computed(() => Math.max(1, Math.ceil(this.total() / this.pageSize())));

  constructor() {
    this.load();
  }

  protected load(): void {
    this.isLoading.set(true);
    this.loadError.set('');
    let params = new HttpParams()
      .set('page', String(this.page()))
      .set('page_size', String(this.pageSize()));
    if (this.actor().trim()) params = params.set('actor', this.actor().trim());
    if (this.entityType()) params = params.set('entity_type', this.entityType());
    if (this.action()) params = params.set('action', this.action());
    if (this.dateFrom()) params = params.set('date_from', this.dateFrom());
    if (this.dateTo()) params = params.set('date_to', this.dateTo());

    this.http.get<AuditListResponse>(this.apiUrl, { params }).subscribe({
      next: (res) => {
        this.rows.set(res.items);
        this.total.set(res.total);
        this.isLoading.set(false);
      },
      error: () => {
        this.loadError.set('Unable to load the audit log.');
        this.isLoading.set(false);
      }
    });
  }

  protected onActorChange(value: string): void { this.actor.set(value); this.page.set(1); this.load(); }
  protected onEntityTypeChange(value: string): void { this.entityType.set(value); this.page.set(1); this.load(); }
  protected onActionChange(value: string): void { this.action.set(value); this.page.set(1); this.load(); }
  protected onDateFromChange(value: string): void { this.dateFrom.set(value); this.page.set(1); this.load(); }
  protected onDateToChange(value: string): void { this.dateTo.set(value); this.page.set(1); this.load(); }

  protected previousPage(): void {
    if (this.page() > 1) { this.page.set(this.page() - 1); this.load(); }
  }
  protected nextPage(): void {
    if (this.page() < this.totalPages()) { this.page.set(this.page() + 1); this.load(); }
  }

  protected formatChanges(row: AuditLogRow): string {
    if (!row.changes || row.changes.length === 0) return '-';
    return row.changes.map((c: AuditChange) => `${c.field}: ${this.fmt(c.old)} → ${this.fmt(c.new)}`).join('; ');
  }

  private fmt(v: unknown): string {
    if (v === null || v === undefined) return '∅';
    return String(v);
  }

  protected formatDate(value: string): string {
    const d = new Date(value);
    return Number.isNaN(d.getTime()) ? value : d.toLocaleString();
  }
}
```
